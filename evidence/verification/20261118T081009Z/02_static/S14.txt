FROM python:3.13-slim@sha256:e48f59cdba69a3cbd9ed2cf0f348d90d45a8d350d98134cf1070bc106ee9ed90

WORKDIR /app

ENV PYTHONDONTWRITEBYTECODE=1 \
    PYTHONUNBUFFERED=1

RUN groupadd -r safi && useradd -r -g safi -d /app -s /sbin/nologin safi

COPY requirements.txt /app/requirements.txt
RUN pip install --no-cache-dir -r /app/requirements.txt

COPY contracts /app/contracts
COPY gate /app/gate
COPY adapters /app/adapters
COPY safichat /app/safichat

RUN mkdir -p /app/data && chown -R safi:safi /app && chmod 750 /app/data

USER safi

EXPOSE 8000

CMD ["uvicorn", "gate.app:app", "--host", "0.0.0.0", "--port", "8000", "--log-config", "/app/gate/logging_config.json"]
